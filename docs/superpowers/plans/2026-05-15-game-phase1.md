```js
  crouch: { row: 2, frames: 3 },

function clamp(v, lo, hi) {
  return v < lo ? lo : v > hi ? hi : v;
}

function getAnimFrame(frameCount, elapsed, fps) {
  return Math.floor(elapsed * fps) % frameCount;

// Returns the leftmost canvas X at which to start drawing tiled parallax layer.
```

```js
// Shape: { image, width, layers: [{ srcYStart, srcYEnd, speed }] }

// ── RENDER ───────────────────────────────────────────────────────
function renderLayer(ctx, img, layer, cameraX, canvasW, canvasH) {
  const bgScale   = canvasH / 1024;
  const scaledBgW = Math.ceil(1536 * bgScale);
  const srcY      = layer.srcYStart * 1024;
  const srcH      = (layer.srcYEnd - layer.srcYStart) * 1024;
  const destY     = layer.srcYStart * canvasH;
  const destH     = (layer.srcYEnd - layer.srcYStart) * canvasH;
  const startX    = getParallaxDrawX(cameraX, layer.speed, scaledBgW);
  for (let x = startX; x < canvasW; x += scaledBgW) {
    ctx.drawImage(img, 0, srcY, 1536, srcH, x, destY, scaledBgW, destH);
  }
}

function renderWorld(ctx, zone, cameraX, canvasW, canvasH) {
  for (const layer of zone.layers) {
    renderLayer(ctx, zone.image, layer, cameraX, canvasW, canvasH);
  }
}
```

```js
// ── PLAYER ───────────────────────────────────────────────────────
function createPlayer(groundY) {
  return {
    x:            200,       // world x — left edge of displayed sprite
    y:            groundY,   // world y — feet (bottom of displayed sprite)
    vy:           0,         // vertical velocity (px/frame at 60fps)
    grounded:     true,
    facingLeft:   false,
    jumpConsumed: false,     // prevents hold-to-multijump
    animState:    'idle',
    animElapsed:  0,         // seconds since current anim started
  };
}

function getAnimState(player, keys) {
  if (!player.grounded)         return 'jump';
  if (keys.down)                return 'crouch';
  if (keys.left || keys.right)  return 'walk';
  return 'idle';
}

function updatePlayer(player, keys, dt, groundY, worldWidth) {
  const dtf = dt * 60;  // normalize physics to 60fps feel

  // Horizontal movement (disabled while crouching)
  let vx = 0;
  if (!keys.down) {
    if (keys.left)  { vx = -WALK_SPEED; player.facingLeft = true;  }
    if (keys.right) { vx =  WALK_SPEED; player.facingLeft = false; }
  }
  player.x = clamp(player.x + vx * dtf, 0, worldWidth - CHAR_W);

  // Jump — one-shot, grounded only, prevents re-trigger while key held
  if (keys.up && player.grounded && !player.jumpConsumed) {
    player.vy           = JUMP_VY;
    player.grounded     = false;
    player.jumpConsumed = true;
  }
  if (!keys.up) player.jumpConsumed = false;

  // Gravity + vertical integration
  if (!player.grounded) {
    player.vy += GRAVITY * dtf;
    player.y  += player.vy * dtf;
  }

  // Ground collision
  if (player.y >= groundY) {
    player.y    = groundY;
    player.vy   = 0;
    player.grounded = true;
  }

  // Animation state
  const next = getAnimState(player, keys);
  if (next !== player.animState) {
    player.animState   = next;
    player.animElapsed = 0;
  } else {
    player.animElapsed += dt;
  }
}
```

```js
function renderPlayer(ctx, spriteImg, player, cameraX, canvasH) {
  const groundY = GROUND_FRAC * canvasH;
  const screenX = Math.round(player.x - cameraX);
  const screenY = Math.round(groundY - CHAR_H);

  // Resolve animation row + current frame
  const animKey  = player.animState === 'idle' ? 'walk' : player.animState;
  const anim     = ANIM[animKey] || ANIM.walk;
  const frame    = player.animState === 'idle'
    ? 0
    : getAnimFrame(anim.frames, player.animElapsed, ANIM_FPS);

  const srcX = frame * FRAME_W;
  const srcY = anim.row * FRAME_H;

  ctx.save();
  if (player.facingLeft) {
    // Mirror: translate to right edge of sprite, flip horizontally, draw at local 0,0
    ctx.translate(screenX + CHAR_W, screenY);
    ctx.scale(-1, 1);
    ctx.drawImage(spriteImg, srcX, srcY, FRAME_W, FRAME_H, 0, 0, CHAR_W, CHAR_H);
  } else {
    ctx.drawImage(spriteImg, srcX, srcY, FRAME_W, FRAME_H, screenX, screenY, CHAR_W, CHAR_H);
  }
  ctx.restore();
}
```

## Task 8: Game loop — wire everything together

- [ ] **Step 1: Replace the entire `init()` function with the full wired-up version**

```js
function init() {
  // Build world
  WORLD = [{
    image: bgImg,
    width: 4096,
    layers: [
      { srcYStart: 0.00, srcYEnd: 0.38, speed: 0.05 },
      { srcYStart: 0.30, srcYEnd: 0.58, speed: 0.20 },
      { srcYStart: 0.55, srcYEnd: 0.77, speed: 0.50 },
      { srcYStart: 0.77, srcYEnd: 1.00, speed: 0.90 },
    ],
  }];

  const groundY = GROUND_FRAC * canvas.height;
  const player  = createPlayer(groundY);
  const camera  = createCamera();
  const zone    = WORLD[0];

  let lastTime = 0;

  function gameLoop(timestamp) {
    const dt = Math.min((timestamp - lastTime) / 1000, 0.05); // cap at 50ms
    lastTime  = timestamp;

    // Update
    updatePlayer(player, keys, dt, groundY, zone.width);
    updateCamera(camera, player, canvas.width, zone.width);

    // Render
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    renderWorld(ctx, zone, camera.x, canvas.width, canvas.height);
    renderPlayer(ctx, spriteImg, player, camera.x, canvas.height);

    requestAnimationFrame(gameLoop);
  }

  // Skip first frame's dt spike
  requestAnimationFrame(ts => { lastTime = ts; requestAnimationFrame(gameLoop); });
```
